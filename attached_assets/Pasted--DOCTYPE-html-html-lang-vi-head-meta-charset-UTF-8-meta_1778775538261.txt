<script>

/* =========================
   DATA
========================= */

let users = [

{

name:"Ynax",

packs:[

{

name:"HSK 2",

language:"chinese",

learned:0,

words:[

{
hanzi:"苹果",
meaning:"quả táo"
},

{
hanzi:"老师",
meaning:"giáo viên"
},

{
hanzi:"学校",
meaning:"trường học"
}

]

}

]

}

];


/* =========================
   LOAD
========================= */

if(localStorage.getItem("flashcardData")){

users =
JSON.parse(
localStorage.getItem("flashcardData")
);

}


/* =========================
   SAVE
========================= */

function saveData(){

localStorage.setItem(
"flashcardData",
JSON.stringify(users)
);

}


/* =========================
   VARIABLES
========================= */

let currentUser = null;
let currentPack = null;
let currentIndex = 0;

let wrongWords = [];
let wrongWordsOnly = [];

let totalCorrect = 0;
let score = 0;

let studyWrongOnly = false;


/* =========================
   USERS
========================= */

function renderUsers(){

let html = "";

users.forEach((user,index)=>{

html += `

<div class="pack">

<h2>
👤 ${user.name}
</h2>

<p>
📚 ${user.packs.length} bộ từ
</p>

<button
class="learn-btn"
onclick="openUser(${index})"
>
Vào học
</button>

<button
class="add-btn"
onclick="editUser(${index})"
>
✏️ Sửa User
</button>

</div>

`;

});

document.getElementById("userGrid")
.innerHTML = html;

}

function createUser(){

let name =
document.getElementById("newUser")
.value
.trim();

if(name === ""){

alert("Nhập tên!");

return;

}

users.push({

name:name,
packs:[]

});

saveData();

document.getElementById("newUser")
.value = "";

renderUsers();

}

function editUser(index){

let user = users[index];

let menu = prompt(

`1 = Đổi tên
2 = Xóa user`

);

if(menu == "1"){

let newName =
prompt(
"Tên mới:",
user.name
);

if(newName){

user.name = newName;

}

}

else if(menu == "2"){

if(confirm("Xóa user?")){

users.splice(index,1);

}

}

saveData();

renderUsers();

}

function openUser(index){

currentUser = users[index];

document.getElementById("dashboard")
.style.display = "none";

document.getElementById("userPage")
.style.display = "block";

document.getElementById("userTitle")
.innerText =
"📚 Bộ từ của "
+ currentUser.name;

renderPacks();

}

function backUsers(){

document.getElementById("dashboard")
.style.display = "block";

document.getElementById("userPage")
.style.display = "none";

renderUsers();

}


/* =========================
   PACKS
========================= */

function createPack(){

let name =
document.getElementById("newPackName")
.value
.trim();

let language =
document.getElementById("packLanguage")
.value;

if(name === ""){

alert("Nhập tên bộ!");

return;

}

currentUser.packs.push({

name:name,
language:language,
learned:0,
words:[]

});

saveData();

document.getElementById("newPackName")
.value = "";

renderPacks();

}

function renderPacks(){

let chineseHTML = "";
let englishHTML = "";

currentUser.packs.forEach((pack,index)=>{

let percent = 0;

if(pack.words.length > 0){

percent =
Math.floor(
(pack.learned / pack.words.length)
* 100
);

}

let card = `

<div class="pack">

<h2>${pack.name}</h2>

<p>
📖 ${pack.words.length} từ
</p>

<p>
✅ ${percent}%
</p>

<div class="progress-bar">

<div
class="progress"
style="width:${percent}%"
></div>

</div>

<button
class="learn-btn"
onclick="openPack(${index})"
>
Học ngay
</button>

<button
class="add-btn"
onclick="editPack(${index})"
>
✏️ Sửa bộ
</button>

</div>

`;

if(pack.language === "english"){

englishHTML += card;

}

else{

chineseHTML += card;

}

});

let both =
chineseHTML !== ""
&&
englishHTML !== "";

if(!both){

document.getElementById("userPackGrid")
.innerHTML = `

<div
style="
display:flex;
justify-content:center;
"
>

<div style="width:80%">

${chineseHTML || englishHTML}

</div>

</div>

`;

}

else{

document.getElementById("userPackGrid")
.innerHTML = `

<div
style="
display:grid;
grid-template-columns:1fr 1fr;
gap:30px;
"
>

<div>

<h2
style="
text-align:center;
color:#ff4f81;
"
>
🇨🇳 Tiếng Trung
</h2>

<div class="pack-grid">

${chineseHTML}

</div>

</div>

<div>

<h2
style="
text-align:center;
color:#2196F3;
"
>
🇬🇧 Tiếng Anh
</h2>

<div class="pack-grid">

${englishHTML}

</div>

</div>

</div>

`;

}

}

function editPack(index){

let pack =
currentUser.packs[index];

let menu = prompt(

`1 = Đổi tên
2 = Xóa bộ`

);

if(menu == "1"){

let newName =
prompt(
"Tên mới:",
pack.name
);

if(newName){

pack.name = newName;

}

}

else if(menu == "2"){

if(confirm("Xóa bộ?")){

currentUser.packs.splice(index,1);

}

}

saveData();

renderPacks();

}


/* =========================
   OPEN PACK
========================= */

function openPack(index){

currentPack =
currentUser.packs[index];

let mode;

if(currentPack.language === "chinese"){

mode = prompt(

`1 = Hiện chữ Hán → viết chữ Hán
2 = Hiện nghĩa → viết chữ Hán`

);

}

else{

mode = prompt(

`1 = Hiện tiếng Anh → nhập nghĩa
2 = Hiện nghĩa → nhập tiếng Anh`

);

}

if(mode == "2"){

currentPack.mode =
"meaning_to_word";

}

else{

currentPack.mode =
"word_to_meaning";

}

currentIndex = 0;

wrongWords = [];

wrongWordsOnly = [];

score = 0;

totalCorrect = 0;

studyWrongOnly = false;

document.getElementById("userPage")
.style.display = "none";

document.getElementById("flashcardPage")
.style.display = "block";

document.getElementById("packTitle")
.innerText =
currentPack.name;

showWord();

}

function backPacks(){

document.getElementById("flashcardPage")
.style.display = "none";

document.getElementById("userPage")
.style.display = "block";

renderPacks();

}


/* =========================
   SHOW WORD
========================= */

function showWord(){

let word;

if(studyWrongOnly){

word =
wrongWordsOnly[currentIndex];

}

else{

word =
currentPack.words[currentIndex];

}

if(!word){

document.getElementById("hanzi")
.innerText = "Không có từ";

return;

}

if(currentPack.mode === "word_to_meaning"){

if(currentPack.language === "english"){

document.getElementById("hanzi")
.innerText = word.word;

}

else{

document.getElementById("hanzi")
.innerText = word.hanzi;

}

}

else{

document.getElementById("hanzi")
.innerText =
word.meaning;

}

document.getElementById("answer")
.value = "";

document.getElementById("result")
.innerText = "";

document.getElementById("meaning")
.innerText = "";

document.getElementById("score")
.innerText = score;

}


/* =========================
   CHECK
========================= */

function checkAnswer(){

let input =
document.getElementById("answer")
.value
.trim()
.toLowerCase();

let word;

if(studyWrongOnly){

word =
wrongWordsOnly[currentIndex];

}

else{

word =
currentPack.words[currentIndex];

}

let correctAnswer = "";

if(currentPack.mode === "word_to_meaning"){

if(currentPack.language === "english"){

correctAnswer =
word.meaning.toLowerCase();

}

else{

correctAnswer =
word.hanzi;

}

}

else{

if(currentPack.language === "english"){

correctAnswer =
word.word.toLowerCase();

}

else{

correctAnswer =
word.hanzi;

}

}

if(input === correctAnswer){

document.getElementById("result")
.innerHTML = "✅ Đúng!";

document.getElementById("meaning")
.innerHTML =
"Đáp án: "
+ correctAnswer;

score++;
totalCorrect++;

if(studyWrongOnly){

wrongWordsOnly.splice(currentIndex,1);

if(wrongWordsOnly.length === 0){

alert(
"🎉 Đã học xong từ sai!"
);

studyWrongOnly = false;

showWord();

return;

}

}

else{

currentPack.learned++;

}

saveData();

setTimeout(()=>{

if(!studyWrongOnly){

currentIndex++;

if(currentIndex >= currentPack.words.length){

currentIndex = 0;

}

}

showWord();

},1000);

}

else{

document.getElementById("result")
.innerHTML = "❌ Sai rồi!";

document.getElementById("meaning")
.innerHTML =
"Đáp án: "
+ correctAnswer;

wrongWords.push(word);

if(studyWrongOnly){

wrongWordsOnly.push(word);

wrongWordsOnly.splice(currentIndex,1);

}

else{

currentPack.words.push(word);

currentPack.words.splice(currentIndex,1);

}

}

}


/* =========================
   ADD WORD
========================= */

function addWordToCurrentPack(){

let wordInput =
document.getElementById("newWord")
.value
.trim();

let meaning =
document.getElementById("newMeaning")
.value
.trim();

if(wordInput === "" || meaning === ""){

alert("Nhập đầy đủ!");

return;

}

if(currentPack.language === "english"){

currentPack.words.push({

word:wordInput,
meaning:meaning

});

}

else{

currentPack.words.push({

hanzi:wordInput,
meaning:meaning

});

}

saveData();

document.getElementById("newWord")
.value = "";

document.getElementById("newMeaning")
.value = "";

alert("Đã thêm từ!");

}


/* =========================
   FINISH
========================= */

function finishStudy(){

let total =
totalCorrect + wrongWords.length;

let text = "";

text +=
"📊 KẾT QUẢ HỌC\n\n";

text +=
"✅ Đúng: "
+ totalCorrect
+ "\n";

text +=
"❌ Sai: "
+ wrongWords.length
+ "\n";

text +=
"📚 Tổng: "
+ total
+ "\n\n";

if(wrongWords.length > 0){

text +=
"⚠️ TỪ ĐÃ SAI:\n\n";

wrongWords.forEach((w,index)=>{

let wordText =
w.hanzi || w.word;

text +=
(index + 1)
+ ". "
+ wordText
+ " = "
+ w.meaning
+ "\n";

});

}

else{

text +=
"🎉 Không có từ sai!";

}

alert(text);

}


/* =========================
   WRONG MODE
========================= */

function startWrongWordsMode(){

if(wrongWords.length === 0){

alert("Không có từ sai!");

return;

}

studyWrongOnly = true;

wrongWordsOnly =
[...wrongWords];

currentIndex = 0;

showWord();

}


/* =========================
   ENTER
========================= */

document
.getElementById("answer")
.addEventListener("keypress",function(e){

if(e.key === "Enter"){

checkAnswer();

}

});


/* =========================
   START
========================= */

renderUsers();
